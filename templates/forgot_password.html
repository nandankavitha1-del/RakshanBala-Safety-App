# ====================================================== # FORGOT PASSWORD PAGE # ====================================================== @app.route("/forgot_password", methods=["GET"]) def forgot_password_page(): return render_template("forgot_password.html") # ====================================================== # RESET PASSWORD # ====================================================== @app.route("/reset_password", methods=["POST"]) def reset_password(): phone = request.form.get("phone", "").strip() new_password = request.form.get("new_password", "") confirm_password = request.form.get("confirm_password", "") phone = phone.replace(" ", "") # Check fields if not phone or not new_password or not confirm_password: return render_template( "forgot_password.html", error="Please fill all fields" ) # Check passwords if new_password != confirm_password: return render_template( "forgot_password.html", error="Passwords do not match" ) # Minimum password length if len(new_password) < 4: return render_template( "forgot_password.html", error="Password must be at least 4 characters" ) # Hash new password password_hash = hashlib.sha256( new_password.encode("utf-8") ).hexdigest() conn = get_database() cursor = conn.cursor() # Check registered phone cursor.execute( "SELECT id FROM users WHERE phone = ?", (phone,) ) user = cursor.fetchone() if not user: conn.close() return render_template( "forgot_password.html", error="Phone number is not registered" ) # Update password cursor.execute(""" UPDATE users SET password = ? WHERE phone = ? """, ( password_hash, phone )) conn.commit() conn.close() # Go back to login return redirect("/login")
